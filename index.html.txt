            if (e.key === 'Escape') closeModal();
        });

        // Live 24fps Camera Timecode
        let frameCount = 0;
        setInterval(() => {
            frameCount = (frameCount + 1) % 24;
            const now = new Date();
            const m = String(now.getMinutes()).padStart(2, '0');
            const s = String(now.getSeconds()).padStart(2, '0');
            const f = String(frameCount).padStart(2, '0');
            const tc = document.getElementById('liveTimecode');
            if (tc) tc.textContent = `00:${m}:${s}:${f}`;
        }, 41);

        // Color Grading Before/After Slider (Zero-shift clip-path geometry)
        const container = document.getElementById('compareContainer');
        const beforeLayer = document.getElementById('beforeLayer');
        const sliderLine = document.getElementById('sliderLine');
        let isDragging = false;

        function moveSlider(clientX) {
            if (!container || !beforeLayer || !sliderLine) return;
            const rect = container.getBoundingClientRect();
            let x = clientX - rect.left;
            x = Math.max(0, Math.min(x, rect.width));
            const percent = (x / rect.width) * 100;
            beforeLayer.style.clipPath = `inset(0 ${100 - percent}% 0 0)`;
            sliderLine.style.left = `${percent}%`;
        }

        if (container) {
            container.addEventListener('mousedown', (e) => {
                isDragging = true;
                moveSlider(e.clientX);
            });
            window.addEventListener('mouseup', () => isDragging = false);
            window.addEventListener('mousemove', (e) => {
                if (isDragging) moveSlider(e.clientX);
            });

            container.addEventListener('touchstart', (e) => {
                isDragging = true;
                moveSlider(e.touches[0].clientX);
            }, { passive: true });
            window.addEventListener('touchend', () => isDragging = false);
            window.addEventListener('touchmove', (e) => {
                if (isDragging) moveSlider(e.touches[0].clientX);
            }, { passive: true });
        }

        // Scroll Reveal Observer
        const revealObserver = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    entry.target.classList.add('active');
                }
            });
        }, { threshold: 0.1 });

        document.querySelectorAll('.reveal').forEach(el => revealObserver.observe(el));

        // Side NLE Timeline Playhead Logic
        (function initSideTimeline() {
            const playhead = document.getElementById('timelinePlayhead');
            const codeLabel = document.getElementById('timelineSectionCode');
            if (!playhead || !codeLabel) return;

            function updateTimeline() {
                const scrollTop = window.scrollY || document.documentElement.scrollTop;
                const docHeight = document.documentElement.scrollHeight - window.innerHeight;
                const progress = docHeight > 0 ? Math.min(1, Math.max(0, scrollTop / docHeight)) : 0;

                // Move playhead along track height
                const track = playhead.parentElement;
                if (track) {
                    const yPos = progress * track.clientHeight;
                    playhead.style.top = `${yPos}px`;
                }

                // Format scroll progress as NLE sequence timecode (00:00 -> 03:00)
                const totalFrames = Math.floor(progress * 180); // 3-section sequence representation
                const sec = String(Math.floor(totalFrames / 60)).padStart(2, '0');
                const frm = String(totalFrames % 60).padStart(2, '0');
                codeLabel.textContent = `${sec}:${frm}`;
            }

            window.addEventListener('scroll', updateTimeline, { passive: true });
            window.addEventListener('resize', updateTimeline);
            updateTimeline();
        })();

        // Interactive Cinema Mouse Trail (Soft Backlight + Centered Focus Ring + Hero Parallax)
        (function initCinemaCursor() {
            const ambient = document.getElementById('cinemaAmbient');
            const glow = document.getElementById('cinemaGlow');
            const ring = document.getElementById('cinemaCursorRing');
            const heroVideo = document.getElementById('heroVideo');

            if (!ambient || !glow || !ring) return;

            let mouseX = window.innerWidth / 2;
            let mouseY = window.innerHeight / 2;
            let glowX = mouseX;
            let glowY = mouseY;
            let ringX = mouseX;
            let ringY = mouseY;
            let isVisible = false;

            window.addEventListener('mousemove', (e) => {
                mouseX = e.clientX;
                mouseY = e.clientY;

                if (!isVisible) {
                    isVisible = true;
                    glowX = mouseX;
                    glowY = mouseY;
                    ringX = mouseX;
                    ringY = mouseY;
                    ambient.style.opacity = '1';
                    ring.style.opacity = '1';
                }

                // Subtle camera pan parallax on Hero video
                if (heroVideo && window.scrollY < window.innerHeight) {
                    const moveX = ((mouseX / window.innerWidth) - 0.5) * -18;
                    const moveY = ((mouseY / window.innerHeight) - 0.5) * -18;
                    heroVideo.style.transform = `scale(1.07) translate(${moveX}px, ${moveY}px)`;
                }
            });

            document.addEventListener('mouseleave', () => {
                isVisible = false;
                ambient.style.opacity = '0';
                ring.style.opacity = '0';
            });

            // Expand focus ring on interactive elements
            const interactiveSelectors = 'a, button, .portfolio-item, #compareContainer';
            document.querySelectorAll(interactiveSelectors).forEach(el => {
                el.addEventListener('mouseenter', () => ring.classList.add('cursor-hover'));
                el.addEventListener('mouseleave', () => ring.classList.remove('cursor-hover'));
            });

            // Smooth interpolation loop (60fps)
            function renderCinemaEffect() {
                // Smooth inertia for soft background light
                glowX += (mouseX - glowX) * 0.08;
                glowY += (mouseY - glowY) * 0.08;

                // Responsive tracking for centered focus ring
                ringX += (mouseX - ringX) * 0.28;
                ringY += (mouseY - ringY) * 0.28;

                glow.style.transform = `translate3d(${glowX}px, ${glowY}px, 0) translate(-50%, -50%)`;
                ring.style.transform = `translate3d(${ringX}px, ${ringY}px, 0) translate(-50%, -50%)`;

                requestAnimationFrame(renderCinemaEffect);
            }

            requestAnimationFrame(renderCinemaEffect);
        })();
    </script>
